<script>
let currentMode="IN TIME";

// Switch IN TIME / DEPARTURE
function switchTab(mode,btn){
  currentMode=mode;
  document.querySelectorAll('.tab').forEach(t=>t.classList.remove('active'));
  btn.classList.add('active');
  document.getElementById('mode').innerText=mode;
}

// Save Data
function saveLive(){
  const data={
    date:new Date().toLocaleDateString(),
    branch: document.getElementById('branchSelect').value,
    mode: currentMode,
    vehicle: vehicle.value,
    weight: weight.value,
    shipments: shipments.value,
    pieces: pieces.value
  };

  // Store in localStorage per branch + mode
  let allData = JSON.parse(localStorage.getItem('mp_live')) || {};
  if(!allData[data.branch]) allData[data.branch]={};
  allData[data.branch][data.mode] = data;
  localStorage.setItem('mp_live', JSON.stringify(allData));

  render(data);
}

// Render Data
function render(d){
  d_date.innerText = d.date || '-';
  d_branch.innerText = d.branch || '-';
  d_vehicle.innerText = d.vehicle || '-';
  d_weight.innerText = d.weight ? d.weight+' kg':'-';
  d_shipments.innerText = d.shipments || '-';
  d_pieces.innerText = d.pieces || '-';
}

// Clock
setInterval(()=>{
  clock.innerText=new Date().toLocaleTimeString();
},1000);

// Load last saved data
let savedAll = JSON.parse(localStorage.getItem('mp_live')) || {};
let lastBranch = Object.keys(savedAll)[0];
if(lastBranch){
  let lastMode = Object.keys(savedAll[lastBranch])[0];
  render(savedAll[lastBranch][lastMode]);
}
</script>
